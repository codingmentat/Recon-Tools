#Paste this into powershell if you cannot run scripts. It expects the service-paths.txt file to be in the same directory

$out=@();Get-CimInstance Win32_Service|%{$p=$_.PathName;if(-not $p -or $p -match 'svchost\.exe'){return};$q=$p.StartsWith('"');if(-not $q -and $p -match '\s' -and $p -match '\\'){$out+="UNQUOTED: $($_.Name) -> $p"};$e=if($q){($p -split '"')[1]}else{($p -split '\s')[0]};try{$d=Split-Path $e -Parent -ErrorAction Stop}catch{return};if(-not $d -or -not(Test-Path $d -PathType Container) -or $d -match 'C:\\Windows\\(System32|SysWOW64|WinSxS)'){return};$t=Join-Path $d "wt_$([guid]::NewGuid()).tmp";try{'x'|Out-File $t -ErrorAction Stop;Remove-Item $t -Force;$out+="WRITABLE: $($_.Name) -> $d ($p)"}catch{}};$out|Tee-Object service-findings.txt;Write-Host "`nTotal findings: $($out.Count)" -ForegroundColor Yellow
